# =============================================================================
# SECTION B: SQL queries (run on downloaded CSV loaded into SQLite/Postgres)
# Table name assumed: inspections
# =============================================================================

-- B1. Column null-rate profile
SELECT COUNT(*) AS total_rows,
       SUM(CASE WHEN facility_type IS NULL OR TRIM(facility_type) = '' THEN 1 ELSE 0 END) AS missing_facility_type,
       SUM(CASE WHEN aka_name IS NULL OR TRIM(aka_name) = '' THEN 1 ELSE 0 END) AS missing_aka_name,
       SUM(CASE WHEN violations IS NULL OR TRIM(violations) = '' THEN 1 ELSE 0 END) AS missing_violations,
       SUM(CASE WHEN latitude IS NULL THEN 1 ELSE 0 END) AS missing_latitude,
       SUM(CASE WHEN risk IS NULL OR TRIM(risk) = '' THEN 1 ELSE 0 END) AS missing_risk
FROM inspections;

-- B2. Results value distribution
SELECT results, COUNT(*) AS cnt,
       ROUND(100.0 * COUNT(*) / SUM(COUNT(*)) OVER (), 2) AS pct
FROM inspections
GROUP BY results
ORDER BY cnt DESC;

-- B3. IC3 violation: Fail must have violations
SELECT inspection_id, dba_name, license_, inspection_date, results
FROM inspections
WHERE results = 'Fail'
  AND (violations IS NULL OR TRIM(violations) = '')
LIMIT 25;

-- B4. Non-operational results (should be excluded from U1)
SELECT results, COUNT(*) AS cnt
FROM inspections
WHERE results IN ('Out of Business', 'No Entry', 'Not Ready', 'Business Not Located')
GROUP BY results
ORDER BY cnt DESC;

-- B5. Duplicate detection on natural key
SELECT license_, inspection_date, results, COUNT(*) AS dup_count
FROM inspections
GROUP BY license_, inspection_date, results
HAVING COUNT(*) > 1
ORDER BY dup_count DESC
LIMIT 25;

-- B6. Risk value anomalies
SELECT risk, COUNT(*) AS cnt
FROM inspections
GROUP BY risk
ORDER BY cnt DESC;

-- B7. U1 scope check: Risk 1 restaurants in 2023-2025
SELECT COUNT(*) AS u1_candidate_rows
FROM inspections
WHERE risk = 'Risk 1 (High)'
  AND facility_type = 'Restaurant'
  AND inspection_date >= '2023-01-01'
  AND inspection_date <= '2025-12-31'
  AND results NOT IN ('Out of Business', 'No Entry', 'Not Ready', 'Business Not Located');

-- B8. NAIVE U1 Q1 (before cleaning — includes duplicates and wrong denominators)
SELECT zip,
       COUNT(*) AS total_inspections,
       SUM(CASE WHEN results = 'Fail' THEN 1 ELSE 0 END) AS failures,
       ROUND(100.0 * SUM(CASE WHEN results = 'Fail' THEN 1 ELSE 0 END) / COUNT(*), 2) AS fail_pct_naive
FROM inspections
WHERE risk = 'Risk 1 (High)'
  AND facility_type = 'Restaurant'
  AND inspection_date >= '2023-01-01'
  AND inspection_date <= '2025-12-31'
GROUP BY zip
ORDER BY fail_pct_naive DESC
LIMIT 10;
